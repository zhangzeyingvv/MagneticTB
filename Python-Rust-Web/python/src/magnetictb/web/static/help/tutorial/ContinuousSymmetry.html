加入精确的连续自旋旋转Add an exact continuous spin rotation

04

先从含 SOC 模型取每个有限操作把参考位置传到目标位置的 6×6 局域块，再追加 C_infty。下面只使用公开不可变对象；orbit → generated site → matrix 的顺序由 DirectProductRepresentation.continuous_site_matrices 明确给出。

Extract the 6-by-6 local block that carries the reference site to its target under each finite operation, then append C_infty. Only public immutable objects are used; DirectProductRepresentation.continuous_site_matrices explicitly follows orbit → generated site → matrix order.

# 提取局域表示、加入连续自旋旋转并初始化新模型# Extract local representations, add a continuous spin rotation, and initialize the new model
def determinant3(matrix):
    return (
        matrix[0][0] * (matrix[1][1] * matrix[2][2] - matrix[1][2] * matrix[2][1])
        - matrix[0][1] * (matrix[1][0] * matrix[2][2] - matrix[1][2] * matrix[2][0])
        + matrix[0][2] * (matrix[1][0] * matrix[2][1] - matrix[1][1] * matrix[2][0])
    )

def cyclotomic_as_exact(value):
    result = 0
    for power, coefficient in enumerate(value.coefficients):
        if coefficient:
            result += coefficient if power == 0 else (
                coefficient * root_of_unity(value.conductor, power)
            )
    return result

local_blocks = []
for operation_index, matrix in enumerate(soc_representation):
    target_site = soc_model.site_actions[0][operation_index][0]
    rows = matrix.rows[6 * target_site:6 * (target_site + 1)]
    local_blocks.append(tuple(
        tuple(cyclotomic_as_exact(value) for value in row[:6])
        for row in rows
    ))

spin_space_operations = []
for label, rotation, translation, parity in finite_operations:
    determinant = determinant3(rotation)
    spin_rotation = tuple(
        tuple(determinant * value for value in row)
        for row in rotation
    )
    spin_space_operations.append(SpinSpaceOperation(
        label, rotation, translation, spin_rotation, parity == "T"
    ))

theta = symbol("theta")
continuous_rotation = (
    (cos(theta), -sin(theta), 0),
    (sin(theta),  cos(theta), 0),
    (0, 0, 1),
)
spin_space_operations.append(SpinSpaceOperation(
    "C_infty", ((1,0,0),(0,1,0),(0,0,1)), (0,0,0),
    continuous_rotation, continuous_parameter=theta,
))

imaginary = exact_complex(0, 1)
spinor_diagonal = (
    exp(-imaginary * theta / 2), exp(imaginary * theta / 2),
) * 3
continuous_matrix = tuple(
    tuple(spinor_diagonal[i] if i == j else 0 for j in range(6))
    for i in range(6)
)
representation = DirectProductRepresentation(
    matrices_by_orbit=(tuple(local_blocks),),
    continuous_site_matrices=((continuous_matrix, continuous_matrix),),
)

initfromrep(
    lattice=lattice,
    lattpar=lattpar,
    wyckoffposition=wyckoff,
    symminformation=tuple(spin_space_operations),
    repinformation=representation,
    orbitalLabels=((
        "pxup", "pxdn", "pyup", "pydn", "pzup", "pzdn",
    ),),
    InitialBondShells=2,
)
no_soc_hamiltonian = hamiltonian_object(1) + hamiltonian_object(2)
print(no_soc_hamiltonian.shape)
print(no_soc_hamiltonian.parameter_names)
