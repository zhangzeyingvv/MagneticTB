from magnetictb import CurrentModelSession

session = CurrentModelSession()
site_orbits = [[[str(value) for value in site] for site in orbit]
               for orbit in session.site_orbits]
print("RepresentationSource", session["RepresentationSource"])
print("RepresentationMode", session["RepresentationMode"])
print("SiteOrbits", site_orbits)
print("LocalBases", [list(basis) for basis in
      session["ModelSpecification", "LocalBases"]])
print("PreparedBondShells",
      [item["Shell"] for item in session["CompiledBondShells"]])
